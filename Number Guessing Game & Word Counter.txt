import random
from collections import Counter

def number_guessing_game():
    print("\n--- Number Guessing Game ---")
    target_number = random.randint(1, 100)
    attempts = 0
    max_attempts = 10
    
    print("I have selected a number between 1 and 100. Can you guess it?")
    
    while attempts < max_attempts:
        try:
            guess = int(input(f"Attempt {attempts + 1}/{max_attempts} - Enter your guess: "))
        except ValueError:
            print("Please enter a valid integer.")
            continue
            
        attempts += 1
        
        if guess < target_number:
            print("Too low! Try a higher number.")
        elif guess > target_number:
            print("Too high! Try a lower number.")
        else:
            score = max(100 - (attempts - 1) * 10, 10)
            print(f"🎉 Congratulations! You guessed the number in {attempts} attempts. Your Score: {score}/100")
            return
            
        if attempts == max_attempts:
            print(f"Game Over! The correct number was {target_number}.")

def word_counter():
    print("\n--- File-Based Word Counter ---")
    filename = input("Enter the name of the text file (e.g., sample.txt): ")
    try:
        with open(filename, 'r', encoding='utf-8') as file:
            text = file.read()
            
        # Tokenize and clean words
        words = text.lower().split()
        words = [w.strip('.,!?;:"()[]{}') for w in words]
        words = [w for w in words if w]  # Remove empty strings
        
        word_counts = Counter(words)
        
        print(f"\nTotal words found: {len(words)}")
        print("Top 5 Most Frequent Words:")
        for word, count in word_counts.most_common(5):
            print(f"  - '{word}': {count} times")
            
    except FileNotFoundError:
        print(f"Error: The file '{filename}' was not found. Please check the file path or create the file first.")

def main():
    while True:
        print("\n=== Python Mini-Projects Menu ===")
        print("1. Play Number Guessing Game")
        print("2. Run File-Based Word Counter")
        print("3. Exit")
        
        choice = input("Enter your choice (1-3): ")
        
        if choice == '1':
            number_guessing_game()
        elif choice == '2':
            word_counter()
        elif choice == '3':
            print("Exiting program. Goodbye!")
            break
        else:
            print("Invalid choice. Please select 1, 2, or 3.")

if __name__ == "__main__":
    main()
